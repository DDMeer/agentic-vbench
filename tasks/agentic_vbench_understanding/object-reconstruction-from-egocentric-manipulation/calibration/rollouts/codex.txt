+import math
+from pathlib import Path

+class Mesh:
+    def __init__(self):
+        self.v = []
+        self.f = []
+
+    def add_v(self, p):
+        self.v.append(tuple(float(x) for x in p))
+        return len(self.v)
+
+    def add_face(self, a, b, c):
+        self.f.append((a, b, c))
+
+    def extend(self, other, offset=(0, 0, 0)):
+        base = len(self.v)
+        ox, oy, oz = offset
+        for x, y, z in other.v:
+            self.add_v((x + ox, y + oy, z + oz))
+        for a, b, c in other.f:
+            self.add_face(a + base, b + base, c + base)
+
+    def write_obj(self, path):
+        with open(path, "w", encoding="ascii") as fh:
+            fh.write("# procedural reconstruction from egocentric clip observations\n")
+            for x, y, z in self.v:
+                fh.write(f"v {x:.6f} {y:.6f} {z:.6f}\n")
+            for a, b, c in self.f:
+                fh.write(f"f {a} {b} {c}\n")

+def cylinder(radius=1.0, height=1.0, seg=64, z0=0.0, cap=True, scale_y=1.0):
+    m = Mesh()
+    bot = []
+    top = []
+    for i in range(seg):
+        t = 2 * math.pi * i / seg
+        x = radius * math.cos(t)
+        y = radius * scale_y * math.sin(t)
+        bot.append(m.add_v((x, y, z0)))
+        top.append(m.add_v((x, y, z0 + height)))
+    for i in range(seg):
+        j = (i + 1) % seg
+        m.add_face(bot[i], bot[j], top[j])
+        m.add_face(bot[i], top[j], top[i])
+    if cap:
+        cb = m.add_v((0, 0, z0))
+        ct = m.add_v((0, 0, z0 + height))
+        for i in range(seg):
+            j = (i + 1) % seg
+            m.add_face(cb, bot[i], bot[j])
+            m.add_face(ct, top[j], top[i])
+    return m

+        top.append(m.add_v((r1 * math.cos(t), r1 * scale_y * math.sin(t), z0 + height)))

+def torus_segment(center=(0, 0, 0), major=1.0, minor=0.08, a0=-1, a1=1, seg=32, tube=12, plane="xz"):
+    m = Mesh()
+    rings = []
+    for i in range(seg + 1):
+        a = a0 + (a1 - a0) * i / seg
+        ca, sa = math.cos(a), math.sin(a)
+        ring = []
+        for k in range(tube):
+            b = 2 * math.pi * k / tube
+            cb, sb = math.cos(b), math.sin(b)
+            rr = major + minor * cb
+            if plane == "xz":
+                p = (center[0] + rr * ca, center[1] + minor * sb, center[2] + rr * sa)
+            else:
+                p = (center[0] + rr * ca, center[1] + rr * sa, center[2] + minor * sb)
+            ring.append(m.add_v(p))
+        rings.append(ring)
+    for i in range(seg):
+        for k in range(tube):
+            l = (k + 1) % tube
+            m.add_face(rings[i][k], rings[i + 1][k], rings[i + 1][l])
+            m.add_face(rings[i][k], rings[i + 1][l], rings[i][l])
+    for ring in (rings[0], rings[-1]):
+        c = m.add_v(tuple(sum(m.v[idx - 1][q] for idx in ring) / tube for q in range(3)))
+        for k in range(tube):
+            m.add_face(c, ring[(k + 1) % tube], ring[k])
+    return m

+def box(x0, x1, y0, y1, z0, z1):
+    m = Mesh()
+    pts = [(x0,y0,z0),(x1,y0,z0),(x1,y1,z0),(x0,y1,z0),

+    for a,b,c,d in quads:

+def extruded_polygon(points, thickness):
+    m = Mesh()
+    z0, z1 = -thickness / 2, thickness / 2
+    bot = [m.add_v((x, y, z0)) for x, y in points]
+    top = [m.add_v((x, y, z1)) for x, y in points]
+    n = len(points)
+    for i in range(n):
+        j = (i + 1) % n
+        m.add_face(bot[i], bot[j], top[j])
+        m.add_face(bot[i], top[j], top[i])
+    # Fan triangulation is fine for the convex/near-convex silhouettes used here.
+    for i in range(1, n - 1):
+        m.add_face(bot[0], bot[i], bot[i + 1])
+        m.add_face(top[0], top[i + 1], top[i])
+    return m

+def rounded_rect(length, width, radius, n=10):
+    pts = []
+    hx, hy = length / 2, width / 2
+    centers = [(hx-radius, hy-radius), (-hx+radius, hy-radius), (-hx+radius, -hy+radius), (hx-radius, -hy+radius)]

+        for i in range(n + 1):
+            a = a0 + (a1 - a0) * i / n
+            pts.append((cx + radius * math.cos(a), cy + radius * math.sin(a)))
+    return pts

+def torus_segment(center=(0, 0, 0), major=1.0, minor=0.08, a0=-1, a1=1, seg=32, tube=12, plane="xz"):
+    m = Mesh()

+def extruded_polygon(points, thickness):

+class Mesh:

+        self.v = []
+        self.f = []

+    def add_v(self, p):
+        self.v.append(tuple(float(x) for x in p))
+        return len(self.v)
+
+    def add_face(self, a, b, c):
+        self.f.append((a, b, c))
+
+    def extend(self, other, offset=(0, 0, 0)):
+        base = len(self.v)
+        ox, oy, oz = offset
+        for x, y, z in other.v:
+            self.add_v((x + ox, y + oy, z + oz))
+        for a, b, c in other.f:
+            self.add_face(a + base, b + base, c + base)
+
+    def write_obj(self, path):
+        with open(path, "w", encoding="ascii") as fh:

+                fh.write(f"v {x:.6f} {y:.6f} {z:.6f}\n")

+def torus_segment(center=(0, 0, 0), major=1.0, minor=0.08, a0=-1, a1=1, seg=32, tube=12, plane="xz"):
